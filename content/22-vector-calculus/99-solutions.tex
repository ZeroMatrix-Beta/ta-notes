\begin{exercisesolution}[Solution to \cref{ex:ai_curl_product_rule_counterexample}]
% Generator: Claude Opus 5 (High)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Take $\varphi \equiv 1$ and $F(x,y,z) := (-y,\,x,\,0)$. The right-hand side is
    $(\grad\varphi)\times F = 0\times F = 0$, since a constant function has vanishing gradient.
    The left-hand side is
    \[\curl(\varphi F) = \curl F
      = \big(\partial_y0 - \partial_z x,\ \partial_z(-y) - \partial_x 0,\
        \partial_x x - \partial_y(-y)\big) = (0,\,0,\,2) \neq 0.\]
    So the two sides differ. The choice needs no work because $\varphi \equiv 1$ kills the
    right-hand side outright while leaving the left-hand side equal to $\curl F$, and any field
    with non-zero curl then finishes the job.
  \item With $\varphi(x,y,z) = z$ and the same $F$, we have $\varphi F = (-yz,\,xz,\,0)$, so
    \[\curl(\varphi F)
      = \big(\partial_y0 - \partial_z(xz),\ \partial_z(-yz) - \partial_x0,\
        \partial_x(xz) - \partial_y(-yz)\big) = (-x,\,-y,\,2z).\]
    The two terms on the right of \cref{prop:curl_product_rule} are
    \[\varphi\,\curl F = z\,(0,0,2) = (0,\,0,\,2z),
      \qquad
      (\grad\varphi)\times F = (0,0,1)\times(-y,x,0) = (-x,\,-y,\,0),\]
    and they add to $(-x,\,-y,\,2z)$, which is $\curl(\varphi F)$.
\end{enumerate}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:ai_which_fields_are_gradients}]
% Generator: Claude Opus 5 (High)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item By \cref{prop:curl_grad_div_curl_zero}, a gradient field has vanishing curl, so a
    non-zero curl rules a field out. Computing both,
    \[\curl F = \big(\partial_y(xy) - \partial_z(xz),\ \partial_z(yz) - \partial_x(xy),\
        \partial_x(xz) - \partial_y(yz)\big) = (x-x,\ y-y,\ z-z) = 0,\]
    \[\curl G = \big(\partial_y0 - \partial_z(-x),\ \partial_zy - \partial_x0,\
        \partial_x(-x) - \partial_yy\big) = (0,\ 0,\ -2) \neq 0.\]
    So $G$ is not a gradient field. Note that this settles $G$ and does \emph{not} yet settle
    $F$: the test is necessary, not by itself sufficient.
  \item Integrating the first component in $x$ suggests $f(x,y,z) := xyz$, and checking all
    three at once,
    \[\nabla f = (yz,\ xz,\ xy) = F.\]
  \item Along $\gamma(t) = (\cos t,\sin t,0)$ we have $\gamma'(t) = (-\sin t,\cos t,0)$ and
    $G(\gamma(t)) = (\sin t,\,-\cos t,\,0)$, so the integrand is
    $\langle G(\gamma),\gamma'\rangle = -\sin^2t - \cos^2t = -1$ and
    \[\int_\gamma G\cdot d\vec s = \int_0^{2\pi}(-1)\,dt = -2\pi.\]
    The curve is closed, $\gamma(2\pi) = \gamma(0)$, so a potential would force the integral to
    be $f(\gamma(2\pi)) - f(\gamma(0)) = 0$ by \cref{prop:work_difference_potential}. Since
    $-2\pi \neq 0$, no potential exists, confirming part \textbf{(a)} without differentiating
    anything.
\end{enumerate}

\begin{remark}[On $\mathbb{R}^3$ the curl test is not merely necessary]
\label{rem:curl_test_sufficient_on_r3}
Part \textbf{(a)} was careful to say that $\curl F = 0$ does not by itself produce a potential,
and in general it does not. On the domain here it does: $\mathbb{R}^3$ is star-shaped, so
\cref{thm:poincare_lemma} applies and every curl-free field on it is a gradient. The guess in
part \textbf{(b)} was therefore guaranteed to succeed before it was made.

What the caution in \textbf{(a)} protects against is the same field on a smaller domain. Delete
the $z$-axis and the conclusion fails, as \cref{q:quiz_q11} exhibits. The hypothesis that
matters is a property of the domain, never of the field.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:differential_identities_true_false}]
% Generator: Claude Opus 5 (High) --- the four statements are the examiner's; the working is
% written for these notes. Verdicts checked afterwards against exercises_2024/mocksol.pdf and
% agree on all four.
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item \textbf{True.} The chain rule applied twice: $\partial_i(e^{w}) = e^{w}\partial_i w$ with
    $w := u^2$, and $\partial_i(u^2) = 2u\,\partial_i u$.
  \item \textbf{True.} Both sides are vectors in $\mathbb{R}^n$, and the $j$-th component of the
    left-hand side is $\partial_i\partial_j u$ while that of the right-hand side is
    $\partial_j\partial_i u$. These agree by \cref{thm:schwarz_mixed_partials}, which applies
    because $u$ is smooth.
  \item \textbf{False.} The quotient rule reads
    $\partial_i(u/v) = (v\,\partial_i u - u\,\partial_i v)/v^2$, and the printed formula is not
    that. It is worth noting that the second summand is not even well defined at a point where
    $\partial_i v$ vanishes, which already settles the matter. For a counterexample where both
    sides do exist, take $n = 1$, $u \equiv 1$ and $v(x) := e^{x}$. Then $u/v = e^{-x}$, so the
    left-hand side is $-e^{-x}$, whereas the right-hand side is $0 + 1/e^{x} = e^{-x}$. The two
    differ in sign.
  \item \textbf{False}, and it fails before any computation: $\Hess u$ is an $n\times n$ matrix,
    so for $n \geq 2$ the right-hand side is not a real number, while the left-hand side is. What
    the product rule for the divergence actually gives is
    \[\divg(u\nabla u) = \nabla u \cdot \nabla u + u\,\divg(\nabla u)
      = \lvert\nabla u\rvert^2 + u\,\Delta u ,\]
    and $\Delta u = \Tr(\Hess u)$ is the \emph{trace} of the Hessian. Replacing $\Hess u$ by
    $\Tr(\Hess u)$ in the printed statement makes it true.
\end{enumerate}

\begin{remark}[The identity in part \textbf{(d)} is worth keeping]
Read from right to left, $\lvert\nabla u\rvert^2 = \divg(u\nabla u) - u\Delta u$ is the
differential half of integration by parts: integrating it over a bounded $C^1$ domain $\Omega$
and applying \cref{thm:gauss_divergence} to the first term turns the left-hand side into a
boundary integral minus $\int_\Omega u\Delta u$. That is Green's first identity, and it is how
$\lvert\nabla u\rvert^2$ is traded for $u\Delta u$ throughout the theory of the Laplacian.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:gradient_field_four_verdicts}]
% Generator: Claude Opus 5 (High) --- the four statements are the examiner's; the working is
% written for these notes. Verdicts checked afterwards against Probeprfg2_Lsg.pdf and agree.
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item \textbf{True.} The condition $\nabla f(x_0) \neq 0$ says exactly that
    $Df_{x_0} : \mathbb{R}^n \to \mathbb{R}$ is surjective, and it persists on a neighbourhood
    $U$ of $x_0$ because $\nabla f$ is continuous. \Cref{thm:regular_value_theorem}, applied to
    $f|_U$ and the value $f(x_0)$, then makes $\{x \in U \mid f(x) = f(x_0)\}$ a submanifold of
    dimension $n-1$. The restriction to a neighbourhood is not a technicality: the full level set
    $\{f = f(x_0)\}$ may contain critical points far from $x_0$ and be no manifold at all.
  \item \textbf{True}, and this is the statement worth remembering. Suppose such a $\gamma$
    existed. Integrating $f$ along it and using the chain rule along a path
    (\cref{sec:chain_rule_along_path}) together with the hypothesis $\gamma' = \nabla f\circ
    \gamma$,
    \[f(\gamma(1)) - f(\gamma(0)) = \int_0^1 \big\langle\nabla f(\gamma(t)),\ \gamma'(t)\big\rangle\,dt
      = \int_0^1 \lvert\gamma'(t)\rvert^2\,dt .\]
    The curve is regular, so $\gamma'$ never vanishes and the integrand is continuous and
    strictly positive; the right-hand side is therefore strictly positive. But $\gamma(0) =
    \gamma(1)$ makes the left-hand side zero. No such curve exists.
  \item \textbf{False.} Take $f(x) := \arctan(x_1)$, so that
    $\nabla f(x) = \big(1/(1+x_1^2),\,0,\dots,0\big)$, which is nowhere zero, while
    $\lvert f\rvert < \pi/2$ everywhere. A non-vanishing gradient forces $f$ to keep increasing
    in some direction, but says nothing about how fast, and a bounded function can increase
    forever.
  \item \textbf{True.} Since $B$ is compact and $f$ continuous, $f|_B$ attains its minimum at
    some $p \in B$. Suppose $p \in \partial B$. Moving inward from $p$ means moving in the
    direction $-\nu(p)$, and the directional derivative there is
    \[D_{-\nu(p)}f(p) = \big\langle\nabla f(p),\ -\nu(p)\big\rangle
      = -\big\langle\nabla f(p),\ \nu(p)\big\rangle < 0\]
    by hypothesis. So $f(p - t\nu(p)) < f(p)$ for all small $t > 0$, and those points lie in $B$
    because $\partial B$ is smooth. That contradicts minimality, so $p$ lies in the interior, and
    a minimum over $B$ attained at an interior point is in particular a local minimum.
\end{enumerate}

\begin{remark}[Two of these are restrictions that only gradients obey]
Parts \textbf{(b)} and \textbf{(d)} both say a gradient field cannot do something, and both are
proved the same way: integrate $f$ rather than reason about $\nabla f$ directly. That is the
whole advantage of being a gradient, and the contrast is sharp. A general vector field has closed
orbits routinely --- $F(x,y) := (-y,x)$ is the rotation field of \cref{ex:autonomous_rotation},
whose orbits are circles --- and part \textbf{(b)} says no gradient field ever does. In physical
terms a conservative field cannot drive a cycle, because going round would return the potential
to its starting value while the field did strictly positive work.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{ex:divergence_under_conjugation}]
% Generator: Claude Opus 5 (High) --- the statement and the one-word hint are the examiner's; the
% proof and the remark are written for these notes.
Write $g := A^{-1}\circ f\circ A$, so that $g(x) = A^{-1}f(Ax)$. A linear map is its own
differential, so the chain rule gives
\[\Jac g(x) = A^{-1}\,\Jac f(Ax)\,A .\]
The divergence of a vector field is the trace of its Jacobian, $\divg g = \sum_i \partial_i g_i =
\Tr(\Jac g)$. Since $\Tr(MN) = \Tr(NM)$, taking $M := A^{-1}\Jac f(Ax)$ and $N := A$,
\[\divg g(x) = \Tr\big(A^{-1}\,\Jac f(Ax)\,A\big) = \Tr\big(\Jac f(Ax)\big) = (\divg f)(Ax) .\]

\begin{remark}[What is really invariant, and how much of it]
The proof used nothing about $\Jac f$ beyond the fact that conjugation does not move the trace,
and the same is true of every coefficient of the characteristic polynomial: $A^{-1}MA$ and $M$
have the same one. So the argument proves more than it was asked to. Not only the divergence but
the whole characteristic polynomial of the Jacobian is unchanged by a linear change of
coordinates, and in particular so is $\det \Jac f$, which is the factor
\cref{thm:change_of_variables} integrates against.

That is worth knowing because the individual summands are \emph{not} invariant. There is no sense
in which $\partial_1 f_1$ survives a change of coordinates; it is only the sum that does, and the
divergence is therefore a property of the vector field rather than of the axes chosen to write it
in. A quantity defined by a formula in coordinates has to earn that status, and this exercise is
the proof that the divergence has.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Solution of \cref{ex:11.1}]
The vector field $V(x,y,z) = (x,y,z-x^2-y^2)$ has divergence
\[\divg V = \partial_x(x) + \partial_y(y) + \partial_z(z-x^2-y^2) = 1+1+1 = 3.\]
$S$ is not itself the boundary of a bounded $C^1$ domain, so close it up with the equatorial
disk $D := \{(x,y,0) : x^2+y^2\leq1\}$: then $\Omega := \{(x,y,z)\in B_1(0) : z\geq0\}$, the
upper half ball, is a bounded $C^1$ domain with $\partial\Omega = S\cup D$, and $S$'s
\qt{inside to outside} normal agrees with the outward normal of $\Omega$ on $S$. By
\cref{thm:gauss_divergence},
\[\int_S V\cdot\nu\,d\vol_2 + \int_D V\cdot\nu_D\,d\vol_2 = \int_\Omega \divg V\,dx
  = 3\vol_3(\Omega) = 3\cdot\frac{2\pi}{3} = 2\pi,\]
using $\vol_3(\Omega) = \frac12\cdot\frac43\pi = \frac{2\pi}{3}$. On $D$ the outward normal is
$\nu_D = -e_3$, and $V(x,y,0) = (x,y,-x^2-y^2)$, so $V\cdot\nu_D = x^2+y^2$; in polar
coordinates,
\[\int_D V\cdot\nu_D\,d\vol_2 = \int_0^{2\pi}\!\!\int_0^1 r^2\cdot r\,dr\,d\varphi
  = 2\pi\left[\frac{r^4}{4}\right]_0^1 = \frac{\pi}{2}.\]
Hence
\[\int_S V\cdot\nu\,d\vol_2 = 2\pi - \frac{\pi}{2} = \boxed{\ \frac{3\pi}{2}\ }.\]
\end{exercisesolution}
