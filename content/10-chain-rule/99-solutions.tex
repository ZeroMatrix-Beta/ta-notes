\begin{exercisesolution}[Solution to \cref{ex:directional_derivative_vanishes}]
\label{sol:directional_derivative_vanishes}
% Generator: Claude Sonnet 5 (Medium)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Define $v(x_2,\dots,x_n) := u(0,x_2,\dots,x_n)$, which is $C^1$ as the restriction of $u$
    to the hyperplane $x_1 = 0$. For fixed $(x_2,\dots,x_n)$, the function
    $t \mapsto u(t,x_2,\dots,x_n)$ has derivative $\partial_1 u \equiv 0$ everywhere, hence is
    constant in $t$; evaluating at $t=0$ gives $u(x_1,\dots,x_n) = u(0,x_2,\dots,x_n) =
    v(x_2,\dots,x_n)$ for all $x_1$. Uniqueness: if some $\tilde v$ also satisfies
    $u(x) = \tilde v(x_2,\dots,x_n)$, then setting $x_1 = 0$ gives $\tilde v = v$.
  \item Since $\nu\cdot e_1 = \nu_1 \neq 0$, consider the invertible linear change of coordinates
    $y_i := x_i - \dfrac{x_1\nu_i}{\nu_1}$ for $i = 2,\dots,n$, together with $x_1$ itself.
    Moving $x_1$ by $t$ while holding $y_2,\dots,y_n$ fixed forces $x_i$ to move by
    $t\nu_i/\nu_1$, i.e.\ moves $x$ in the direction $\tfrac{1}{\nu_1}\nu$. By the hint, applying
    the mean value theorem to $u(x+t\nu)$ shows $t \mapsto u(x+t\nu)$ is constant since
    $\partial_\nu u \equiv 0$; consequently $u$, expressed in the coordinates $(x_1, y_2,\dots,
    y_n)$, has vanishing $x_1$-derivative and is therefore (by part \textbf{(a)}) independent of
    $x_1$ in these coordinates. Hence, there is a (unique, by the same argument as in part
    \textbf{(a)}) function $w \in C^1(\mathbb{R}^{n-1})$ with
    \[
    u(x_1,\dots,x_n) = w\!\left(x_2 - \frac{x_1\nu_2}{\nu_1},\dots,x_n - \frac{x_1\nu_n}{\nu_1}\right).
    \]
  \item In general, only \textbf{locally}: on any open subset where every fiber
    $\{x_1 : (x_1,x_2,\dots,x_n) \in U\}$ (for fixed $(x_2,\dots,x_n)$) is connected, the same
    argument as in part \textbf{(a)} shows $u$ is locally of the form $v(x_2,\dots,x_n)$. But this
    need not hold \textbf{globally} on all of $U$, even though $U$ itself is connected: with
    $U := \{(x,y) : x^2 < y < x^2+1\}$ and $u(x,y) := \max\{0,y-2\}^2$ for $x \geq 0$, $u := 0$
    for $x \leq 0$, one checks $\partial_1 u \equiv 0$ throughout $U$ (near $x=0$ both branches
    agree, since $y < 2$ there), yet $u$ is not globally independent of $x$: for $y > 2$, the
    fiber of $U$ over $y$ splits into two disconnected intervals (one with $x>0$, one with
    $x<0$), and $u$ takes different values on each. So, connectedness of $U$ alone does not
    suffice; what is needed is connectedness of each vertical fiber of $U$.
\end{enumerate}
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{ex:ai_directional_derivative}]
\label{sol:ai_directional_derivative}
% Generator: Gemini 3.6 Flash (Medium)
The gradient of $f(x,y)$ is $\nabla f(x,y) = (2xy, x^2)\transp$. Evaluating at $(1,2)$ gives $\nabla f(1,2) = (4,1)\transp$. Since $f$ is continuously differentiable ($C^\infty$), the directional derivative along the unit vector $v$ is given by the inner product:
\[
D_v f(1,2) = \nabla f(1,2) \cdot v = \begin{pmatrix} 4 \\ 1 \end{pmatrix} \cdot \frac{1}{\sqrt{2}}\begin{pmatrix} 1 \\ 1 \end{pmatrix} = \frac{4 + 1}{\sqrt{2}} = \frac{5}{\sqrt{2}}.
\]
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:directional_derivatives_computation}]
\label{sol:directional_derivatives_computation}
% Generator: Claude Sonnet 5 (Medium)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Since $e^{-x^2-y^2} > 0$ everywhere, the gradient of $f(x,y) = xy^2e^{-x^2-y^2}$ vanishes
    exactly where its two factors below vanish:
    \[
    \partial_x f = y^2(1-2x^2)e^{-x^2-y^2}, \qquad \partial_y f = 2xy(1-y^2)e^{-x^2-y^2}.
    \]
    \textbf{If $y=0$:} both partials vanish for every $x$ (indeed $f(x,0) \equiv 0$), so the
    entire line $\{(x,0) : x \in \mathbb{R}\}$ consists of critical points.
    \textbf{If $y \neq 0$:} $\partial_x f = 0$ forces $x = \pm\tfrac{1}{\sqrt2}$, and since then
    $x \neq 0$, $\partial_y f = 0$ forces $y = \pm 1$. This gives the four additional critical
    points $\big(\pm\tfrac{1}{\sqrt2}, \pm 1\big)$.
  \item A directional derivative $\partial_v g(x,y) = \nabla g(x,y)\cdot v$ is always
    \textbf{linear} in $v$. The prescribed expression
    $2\cos(x^2y)xv_2 + \cos(x^2y)v_1^2 y$ contains a $v_1^2$ term with coefficient
    $y\cos(x^2y)$, which is not identically zero (e.g.\ at $(x,y)=(0,1)$ it equals $1$). Hence
    the expression is not linear in $v$ at every point, so \textbf{no such $g \in C^1(\mathbb{R}^2)$
    can exist}.
\end{enumerate}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:derivatives_gradient_polar_computation}]
\label{sol:derivatives_gradient_polar_computation}
% Generator: Claude Opus 5 (High)
Throughout, $v$ is the direction and $x$ the base point. Where $f$ is differentiable the answer is
just $\partial_v f(x) = \langle\nabla f(x), v\rangle$; the interest is in the points where it is
not, which is what the second half of each part is asking about.
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item On $\mathbb{R}^2\setminus\{x_2=0\}$ both functions are smooth, so
    \[\partial_v\!\left(\frac{x_1}{x_2}\right) = \frac{v_1}{x_2} - \frac{x_1v_2}{x_2^2},
      \qquad
      \partial_v\!\left(e^{-x_1/x_2}\right)
        = -e^{-x_1/x_2}\left(\frac{v_1}{x_2} - \frac{x_1v_2}{x_2^2}\right).\]
    Neither extends continuously across $\{x_2 = 0\}$. For $x_1/x_2$, fix $x_1 = 1$ and let
    $x_2 \to 0^{\pm}$: the values run off to $\pm\infty$. For $e^{-x_1/x_2}$ the failure is more
    interesting, since the function is \emph{bounded} on one side: with $x_1 = 1$ fixed,
    $x_2 \to 0^+$ sends $-1/x_2 \to -\infty$ and the value to $0$, while $x_2 \to 0^-$ sends it
    to $+\infty$. A limit existing from one side is not enough.
  \item The first function is smooth on all of $\mathbb{R}^2$, with
    \begin{align*}
      \partial_v\!\left(e^{x_1x_2}\sin(x_1{+}x_2^2)\right)
        &= e^{x_1x_2}\big(x_2\sin(x_1{+}x_2^2) + \cos(x_1{+}x_2^2)\big)\,v_1 \\
        &\quad + e^{x_1x_2}\big(x_1\sin(x_1{+}x_2^2) + 2x_2\cos(x_1{+}x_2^2)\big)\,v_2,
    \end{align*}
    and nothing needs extending. The other three are the point of the exercise, and they separate
    three different behaviours at the origin.
    \begin{itemize}
      \item $\dfrac{x_1^2}{x_1^2+x_2^2}$ is homogeneous of degree $0$, with
        \[\partial_v f(x) = \frac{2x_1x_2\,(x_2v_1 - x_1v_2)}{(x_1^2+x_2^2)^2}.\]
        It has \textbf{no continuous extension}: it equals $1$ on $\{x_2 = 0\}$ and $0$ on
        $\{x_1 = 0\}$.
      \item $\dfrac{x_1^2x_2}{x_1^2+x_2^2}$ \textbf{does} extend continuously by $f(0):=0$, since
        $x_1^2 \leq x_1^2+x_2^2$ gives $|f| \leq |x_2|$. It is not differentiable there:
        \[\partial_vf(0) = \lim_{t\to0}\frac{f(tv)}{t}
          = \lim_{t\to0}\frac{t^3v_1^2v_2}{t^3(v_1^2+v_2^2)} = \frac{v_1^2v_2}{v_1^2+v_2^2},\]
        which exists in every direction and is \emph{not linear} in $v$.
      \item $\dfrac{x_1^2x_2}{x_1^2+x_2^4}$ also satisfies $|f| \leq |x_2|$, so it too extends
        continuously by $0$. Here
        \[\partial_vf(0) = \lim_{t\to0}\frac{t^3v_1^2v_2}{t^3\,(v_1^2 + t^2v_2^4)}
          = \begin{cases} v_2 & v_1 \neq 0,\\ 0 & v_1 = 0,\end{cases}\]
        which is not even continuous in $v$, let alone linear. Again continuous, not
        differentiable.
    \end{itemize}
    All three are instances of \cref{ex:all_directional_derivatives_exist}: every directional
    derivative can exist without the map $v \mapsto \partial_vf(0)$ being linear, and linearity
    is exactly what differentiability demands.
  \item Here $x \in \mathbb{R}^n\setminus\{0\}$ for the first four, $e \in \mathbb{R}^n$ and
    $g \in C^1(\mathbb{R})$:
    \[\partial_v|x| = \frac{\langle x,v\rangle}{|x|}, \qquad
      \partial_v|x|^\alpha = \alpha\,|x|^{\alpha-2}\langle x,v\rangle, \qquad
      \partial_v\,g(|x|) = g'(|x|)\,\frac{\langle x,v\rangle}{|x|},\]
    \[\partial_v\!\left(\frac{x}{|x|}\right)
        = \frac{1}{|x|}\left(v - \frac{\langle x,v\rangle}{|x|^2}\,x\right), \qquad
      \partial_v\,(x\cdot e) = \langle e,v\rangle, \qquad
      \partial_v\,g(x\cdot e) = g'(\langle x,e\rangle)\,\langle e,v\rangle.\]
    The fourth is worth reading geometrically: $x/|x|$ records only a direction, so moving $x$
    along itself changes nothing, and indeed the bracket is exactly $v$ with its component along
    $x$ removed.
  \item For matrix arguments the cleanest route is the one Corsin's hint suggests: expand
    $f(x+tv)$ in powers of $t$ and read off the coefficient of $t$. The first three maps are
    linear, hence their own derivatives:
    \[\partial_v(ax) = av, \qquad \partial_v(x\transp) = v\transp, \qquad
      \partial_v \Tr(x) = \Tr(v).\]
    For the rest, $(x+tv)^2 = x^2 + t(xv+vx) + t^2v^2$ and
    $(x+tv)^3 = x^3 + t(x^2v + xvx + vx^2) + O(t^2)$, so
    \[\partial_v(x^2) = xv + vx, \qquad \partial_v(x^3) = x^2v + xvx + vx^2, \qquad
      \partial_v\Tr(x^2) = \Tr(xv+vx) = 2\Tr(xv),\]
    the last using cyclicity of the trace. Note that $\partial_v(x^2) \neq 2xv$: matrices do not
    commute, and the familiar one-variable answer is precisely what fails.
  \item \textbf{(*)} Differentiate the identity $(x+tv)(x+tv)^{-1} = \id$ at $t=0$. The product
    rule gives $v\,x^{-1} + x\,\partial_v(x^{-1}) = 0$, hence
    \[\partial_v(x^{-1}) = -\,x^{-1}v\,x^{-1}.\]
    Writing $x^{-2} = x^{-1}x^{-1}$ and applying the product rule again,
    \[\partial_v(x^{-2}) = -\,x^{-1}v\,x^{-2} \;-\; x^{-2}v\,x^{-1}.\]
    Both reduce to the scalar formulas $-v/x^2$ and $-2v/x^3$ when $n=1$; for $n>1$ the two terms
    genuinely differ and cannot be combined.
\end{enumerate}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:curve_chain_rule_car_earth}]
\label{sol:curve_chain_rule_car_earth}
% Generator: Claude Opus 5 (High)
Write $\theta(t) := \omega t + a(t)$ for the longitude actually occupied in space and
$\varphi(t) := b(t)$ for the latitude, so that
\[r(t) = R\big(\cos\varphi\cos\theta,\ \cos\varphi\sin\theta,\ \sin\varphi\big).\]
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item The computation is far shorter in the moving orthonormal frame
    \begin{align*}
      e_r      &:= (\cos\varphi\cos\theta,\ \cos\varphi\sin\theta,\ \sin\varphi), \\
      e_\varphi&:= (-\sin\varphi\cos\theta,\ -\sin\varphi\sin\theta,\ \cos\varphi), \\
      e_\theta &:= (-\sin\theta,\ \cos\theta,\ 0),
    \end{align*}
    in which $r = R\,e_r$. Differentiating the three frame vectors by the chain rule gives
    \[\dot e_r = \varphi'\,e_\varphi + \theta'\cos\varphi\;e_\theta, \qquad
      \dot e_\varphi = -\varphi'\,e_r - \theta'\sin\varphi\;e_\theta, \qquad
      \dot e_\theta = \theta'\big({-}\cos\varphi\,e_r + \sin\varphi\,e_\varphi\big).\]
    Hence $r' = R\big(\varphi'\,e_\varphi + \theta'\cos\varphi\,e_\theta\big)$, and
    differentiating once more and collecting the three components,
    \begin{align*}
      r''(t) = R\Big[&-\big(\varphi'^2 + \theta'^2\cos^2\varphi\big)\,e_r \\
        &+ \big(\varphi'' + \theta'^2\sin\varphi\cos\varphi\big)\,e_\varphi \\
        &+ \big(\theta''\cos\varphi - 2\,\theta'\varphi'\sin\varphi\big)\,e_\theta\Big].
    \end{align*}
    The first component is the centripetal term; the other two are tangential.
  \item Substituting $\varphi = b$, $\theta' = \omega + a'$ and $\theta'' = a''$, the assumption
    $r''(t) = \lambda(t)\,r(t) = \lambda(t)R\,e_r$ says precisely that the $e_\varphi$ and
    $e_\theta$ components vanish. The frame is orthonormal, so there is no cancellation between
    them to hope for. That is
    \[b'' + \sin b\cos b\,(\omega+a')^2 = 0, \qquad
      \cos b\;a'' - 2\sin b\,(\omega+a')\,b' = 0,\]
    which is the claim. Away from the poles $\cos b \neq 0$, so dividing the second equation by
    $\cos b$ gives
    \[a'' - 2\tan b\,(\omega+a')\,b' = 0.\]
\end{enumerate}

% Reclassified from ainote to remark: two observations about the equations themselves.
\begin{remark}[Reading the two equations physically]
Two features are worth noticing. First, $\omega$ enters only through the combination
$\omega + a'$, which is the \emph{absolute} angular velocity. That is what one would hope for,
since the equations ought not to distinguish a car driving east on a still Earth from a parked car
on a rotating one. Second, the factor $2$ in the $e_\theta$ equation is the Coriolis term, and it
is the same $2\theta'\varphi'$ that appears in the acceleration in polar coordinates one dimension
lower.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{ex:examHS24_TF_chain_rule}]
\label{sol:chain_rule_higher_order_tf}
% Generator: Gemini 3.1 Pro (High)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item \textbf{True:} This is the standard chain rule $\nabla(u \circ F) = \nabla u(F) \cdot \Jac F$, where the $i$-th component (derivative with respect to $x_i$) is the product of $\nabla u(F)$ with the $i$-th column of the Jacobian $\Jac F$.
  \item \textbf{False:} Applying the product rule to the expression from part \textbf{(a)} yields $\partial_{ii}(u \circ F) = \partial_i \big( \sum_{j=1}^n (\partial_j u \circ F) \, \partial_iF_j \big) = \sum_{j=1}^n \big( \partial_i(\partial_j u \circ F) \, \partial_iF_j + (\partial_j u \circ F) \, \partial_{ii}F_j \big)$. Expanding the first term with the chain rule again gives $\sum_{j=1}^n \big( \sum_{k=1}^n (\partial_{jk} u \circ F) \, \partial_iF_k \big) \partial_iF_j$. The given formula erroneously collapses the double sum $\sum_{j,k}$ into a single sum $\sum_{j} \partial_{jj} u (\partial_iF_j)^2$, missing all the cross terms where $j \neq k$.
  \item \textbf{True:} Since $F$ and $u$ are both $C^2$, their composition $u \circ F$ is also $C^2$. By Schwarz's theorem, the mixed partial derivatives commute.
\end{enumerate}
\end{exercisesolution}

\begin{exercisesolution}[Solution of \cref{ex:regularity_ladder_xy2}]
\label{sol:regularity_ladder_xy2}
% Generator: Claude Opus 5 (High)
One substitution settles everything. In polar coordinates $x = r\cos\theta$, $y = r\sin\theta$,
\[f(r\cos\theta,\, r\sin\theta) = \frac{r\cos\theta \cdot r^2\sin^2\theta}{r^2}
  = r\,\cos\theta\sin^2\theta \qquad (r > 0),\]
so $f$ is positively homogeneous of degree $1$: it grows linearly along every ray, with a
ray-dependent slope. That single formula answers \textbf{(c)} and \textbf{(d)} at once.
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item \textbf{True.} Away from the origin $f$ is a quotient of polynomials whose denominator does
    not vanish, and such a quotient is $C^\infty$.
  \item \textbf{True.} Along the $x$-axis $f(x,0) = 0$ for every $x$, so
    $\partial_xf(0,0) = \lim_{s\to0}\big(f(s,0)-f(0,0)\big)/s = 0$. The same computation on the
    $y$-axis gives $\partial_yf(0,0) = 0$, so both partial derivatives exist and vanish.
  \item \textbf{True.} From the polar formula,
    $\lvert f\rvert = r\,\lvert\cos\theta\rvert\sin^2\theta \leq r$, and the bound is uniform in
    $\theta$. Consequently $f(x,y) \to 0 = f(0,0)$ as $(x,y) \to (0,0)$, and $f$ is continuous
    there. Continuity away from the origin is \textbf{(a)}.
  \item \textbf{False}, and $f$ is not even differentiable at the origin. By \textbf{(b)} the only
    candidate for the differential is $L = 0$, so \cref{rem:differentiability_recipe} asks whether
    $f(h)/\lvert h\rvert \to 0$. In polar coordinates that quotient is
    \[\frac{f(r\cos\theta, r\sin\theta)}{r} = \cos\theta\sin^2\theta,\]
    which does not depend on $r$ at all. It is $0$ along the axes but
    $\tfrac{1}{2\sqrt2} \neq 0$ along $\theta = \pi/4$, so the limit is not $0$ and $f$ is not
    differentiable at $(0,0)$. Since differentiability is weaker than $C^1$
    (\cref{thm:continuous_partials_imply_differentiable} and the chain of implications above it),
    $f \notin C^1(\mathbb{R}^2)$.
\end{enumerate}

\begin{remark}[Where this function sits in the chain]
The four verdicts place $f$ at a definite height in the diagram of
\cref{sec:sufficient_condition_differentiability}: strictly above $\tfrac{xy}{x^2+y^2}$, which
fails already at continuity, and strictly below $\tfrac{x^2y^2}{\sqrt{x^2+y^2}}$, which is
differentiable. Degree-$1$ homogeneity is what pins it there, and it does so in general. Such an
$f$ is automatically continuous at the origin, since $\lvert f\rvert \leq Cr$; and it is
differentiable there only if it is already \emph{linear}, because $f(h)/\lvert h\rvert$ then
depends on the direction alone, so it can converge only by agreeing with $L(h)/\lvert h\rvert$
exactly, in every direction at once.
\end{remark}
