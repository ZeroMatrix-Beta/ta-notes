\section{The mid-semester quiz}
\label{sec:repetition_quiz}

\begin{ainote}
Corsin credits the questions to Prof.\ Serra and Prof.\ Lang, and ran them from a shared Kahoot
at the mid-semester repetition session. On his pages, \textcolor{green!60!black}{green} marks the
\emph{correct} statement and \textcolor{red!80!black}{red} an \emph{incorrect} one; that colouring
is rendered below as explicit \textbf{True}/\textbf{False} verdicts.
\end{ainote}

\begin{question}
\label{q:quiz_q1}
Let $U := \{(x,y) \in \mathbb{R}^2 : 1 \leq x^2+y^2 \leq 4\}$ and
$f(x,y) := \sin(xy) - y^4$. Which of the following holds?
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item $f(U) = [a,b]$ for some $a < b$.
  \item $f(U)$ is disconnected and closed with two connected components.
\end{enumerate}
\end{question}

\begin{answer}
\textbf{(a) is true, (b) is false.} $U$ is compact and connected and $f$ is continuous, so
$f(U) \subseteq \mathbb{R}$ is compact (\cref{item:continuity_preserves_compact}) and connected
(\cref{thm:continuity_preserves_connected}). The connected subsets of $\mathbb{R}$ are exactly
the intervals, and a compact one is closed and bounded, so $f(U) = [a,b]$.
\end{answer}

\begin{ainote}
Worth noting how little computation this needs: $f$ is never evaluated. The annulus $U$ is
connected (it is \emph{not} simply connected, but that is a different property and irrelevant
here), and the two preservation theorems do the rest. Recognising that a question
is asking about preserved properties rather than about values is most of the work.
\end{ainote}

\begin{question}
\label{q:quiz_q2}
Let $X \subseteq \mathbb{R}^2$ with $X \neq \mathbb{R}^2$ and $X \neq \emptyset$. If $X$ is
complete, is it closed?
\end{question}

\begin{answer}
\textbf{True.} Let $(a_n)_{n=0}^\infty$ be a convergent sequence with elements in $X$. Then
$(a_n)$ is Cauchy, and by completeness it has a limit point in $X$. So $X$ is sequentially
closed, hence closed (\cref{prop:open_closed_via_sequences}).
\end{answer}

\begin{ainote}
The limit is taken in $\mathbb{R}^2$, and completeness of $X$ is what forces it back into $X$.
The converse also holds here, but only because the ambient space $\mathbb{R}^2$ is itself
complete: a closed subset of a complete space is complete. In a non-complete ambient space
closedness gives nothing. Note that $\mathbb{Q}$ is closed in $\mathbb{Q}$ and not complete.
\end{ainote}

% Source: Corsin Nick/Class Notes/Week 8.pdf, p. 3
\begin{question}
\label{q:quiz_q3}
Let $X \subseteq \mathbb{R}^2$ as above. If $X$ is not open, is it closed?
\end{question}

\begin{answer}
\textbf{False.}
\end{answer}

% Generator: Claude Opus 5 (Medium) --- Corsin gives the verdict without a counterexample.
\begin{ainote}
Corsin marks this false but gives no counterexample; the half-open square
$X := [0,1] \times (0,1)$ from \cref{sec:open_closed_sets} is one, being neither open
nor closed. \qt{Open} and \qt{closed} are not opposites and not exhaustive: $\emptyset$ and
$\mathbb{R}^2$ are both, most sets are neither, and the excluded values
$X \neq \emptyset,\ X \neq \mathbb{R}^2$ in the hypothesis are there precisely to rule out the
two clopen cases.
\end{ainote}

\begin{question}
\label{q:quiz_q4}
Let
\[f(x,y,z) := \frac{xy^2}{x^2+y^2+z^2} \quad \text{for } (x,y,z) \neq 0, \qquad f(0,0,0) := 0.\]
Is $f \in C^1(\mathbb{R}^3)$?
\end{question}

\begin{answer}
\textbf{False.} We compute
\[\partial_xf = \frac{y^2}{x^2+y^2+z^2} - 2\,\frac{x^2y^2}{(x^2+y^2+z^2)^2}.\]
In spherical coordinates $x = r\sin\theta\cos\varphi$, $y = r\sin\theta\sin\varphi$,
$z = r\cos\theta$, every power of $r$ cancels:
\[\partial_xf = \sin^2\theta\sin^2\varphi - 2\sin^4\theta\cos^2\varphi\sin^2\varphi.\]
The limit $\lim_{r\to0}\partial_xf(r,\theta,\varphi)$ is therefore ill-defined, since it depends
on the direction $(\theta,\varphi)$ along which the origin is approached. Consequently
$\partial_xf$ is not sequentially continuous at $0$.
\end{answer}

\begin{ainote}
The cancellation of $r$ is the whole argument, and it is worth naming the reason: $\partial_xf$
is \newterm{homogeneous of degree $0$}, so it is constant along each ray from the origin and
takes a different constant on different rays. Compare
\Cref{ex:all_directional_derivatives_exist}, where the same trick (restrict to a ray, watch the
parameter cancel) exposed a function whose directional derivatives all existed without it being
differentiable.
\end{ainote}

% Source: Corsin Nick/Class Notes/Week 8.pdf, pp. 3--4
\begin{question}
\label{q:quiz_q5}
Let $f \in C^\infty(\mathbb{R}^3)$ and $\alpha \in \mathbb{R}$ be such that
\[\nabla f(0) = (0,0,0), \qquad
\Hess f(0) = \begin{pmatrix} 1 & \alpha & 0 \\ \alpha & 2 & 0 \\ 0 & 0 & 1\end{pmatrix}.\]
Which of the following is \textbf{false}?
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item For all $\alpha \in \mathbb{R}$, $0$ is not a local maximum point of $f$.
  \item For $\alpha > \sqrt2$, $0$ is a saddle point.
  \item For $\alpha \neq \pm\sqrt2$, $\nabla f : \mathbb{R}^3 \to \mathbb{R}^3$ is a local
    diffeomorphism around $0$.
  \item For $\alpha < \sqrt2$, $0$ is a local minimum point of $f$.
\end{enumerate}
\end{question}

\begin{answer}
\textbf{(d) is the false one.}

\textbf{(a) is true,} since $1 > 0$ is clearly an eigenvalue of $\Hess f(0)$ for any choice of
$\alpha$ (the third standard basis vector is an eigenvector), so $\Hess f(0)$ is never negative
semidefinite and $0$ can never be a local maximum.

\textbf{(b) is true.} Let $\lambda_1,\lambda_2,\lambda_3 = 1$ be the eigenvalues of
$\Hess f(0)$. Then
\[\lambda_1\lambda_2 = \det\big(\Hess f(0)\big) = 2-\alpha^2 < 0 \quad \text{for } \alpha > \sqrt2,\]
so $\Hess f(0)$ has both positive and negative eigenvalues, making it indefinite, and $0$ is a
saddle point by \cref{thm:hessian_test}.

\textbf{(c) is true.} For these values $\det(\Hess f(0)) \neq 0$, and therefore the inverse
function theorem (\cref{thm:inverse_function_theorem}) applies. Note that
$\Hess f(0) = D(\nabla f)_0$: the Hessian of $f$ \emph{is} the differential of the gradient map.

\textbf{(d) is false,} because the same reasoning as in \textbf{(b)} gives that for
$\alpha < -\sqrt2 < \sqrt2$ we again have $2-\alpha^2 < 0$, so $0$ is a saddle point of $f$, not
a local minimum. The statement is true only on $-\sqrt2 < \alpha < \sqrt2$.
\end{answer}

\begin{ainote}
Corsin's third bullet reads \qt{for $\alpha \neq \sqrt2$}, but $\det(\Hess f(0)) = 2-\alpha^2$
vanishes at $\alpha = -\sqrt2$ as well, so it must read $\alpha \neq \pm\sqrt2$; typeset with
$\pm$ above. The oversight is a conspicuous one here, since the \emph{next} bullet is false for
exactly the reason the third one omits. The quiz is built around the missing negative root.

\end{ainote}

\begin{question}
\label{q:quiz_q6}
Let $V := \{(x,y) \in \mathbb{R}^2 : y^4+y^2 = x^3+x\}$. Decide:
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item $V$ is a graph over $y$ in a neighbourhood of $(0,0)$.
  \item $V$ is a graph over $x$ in a neighbourhood of $(0,0)$.
\end{enumerate}
\end{question}

\begin{answer}
\textbf{(a) True.} Notice that $V = f^{-1}\{0\}$ for $f(x,y) := y^4+y^2-x^3-x$. Then
\[\Jac f(x,y) = (\partial_xf,\ \partial_yf) = (-3x^2-1,\ 4y^3+2y), \qquad \Jac f(0,0) = (-1,\ 0).\]
Since $\partial_xf(0,0) = -1 \neq 0$, the implicit function theorem
(\cref{thm:implicit_function_theorem}) applies with $x$ as the variable to solve for: in a
neighbourhood of $(0,0)$,
\[(x,y) \in V \iff f(x,y) = 0 \iff x = x(y).\]
The theorem does \textbf{not} apply for $y$, since $\partial_yf(0,0) = 0$.

\textbf{(b) False.} Solving the defining equation for $y^2$ by the quadratic formula gives
\[y^2 = -\tfrac12 + \sqrt{\tfrac14 + x^3 + x},\]
which has no solution for $x \in (-\varepsilon,0)$: there $x^3+x < 0$, so the square root is
smaller than $\tfrac12$ and the right-hand side is negative, while $y^2 \geq 0$. So $V$ contains
no points at all just to the left of the origin, and cannot be a graph over $x$ there.
\end{answer}

\begin{ainote}
The two parts are settled by different means, and the asymmetry is the point. Part \textbf{(a)}
is decided by \emph{checking a derivative}; part \textbf{(b)} needs an \emph{explicit
computation} showing the set is empty on one side. That is because the implicit function theorem
yields no negative results: failure of its hypothesis proves nothing on its own, exactly as
discussed after \cref{thm:implicit_function_theorem}.
\end{ainote}

\begin{question}
\label{q:quiz_q7}
Let $\gamma : I \to \mathbb{R}^2$ be injective and smooth, with $I$ an open interval. Is
$\gamma(I)$ then a smooth submanifold of $\mathbb{R}^2$?
\end{question}

\begin{answer}
\textbf{False.} Consider
\[\gamma : (0,2\pi) \to \mathbb{R}^2, \qquad t \mapsto \big(\sin t,\ \sin 2t\big),\]
whose image is a \newterm{figure eight} \germanfor{Lemniskate}. This $\gamma$ is injective and
smooth, but there is no graphical representation of $\operatorname{im}(\gamma)$ around $(0,0)$.
\end{answer}

\begin{ainote}
Two things are worth extracting. First, how injectivity survives a self-crossing at all: the two
parameter ends $t \to 0^+$ and $t \to 2\pi^-$ both send $\gamma(t) \to (0,0)$, but neither
\emph{attains} it, so the image passes through the origin only once (via $t = \pi$) while
being approached from two further directions.

Second, this locates precisely which hypothesis of \cref{thm:local_parametrization} fails. It is
not injectivity, and not injectivity of $D\gamma_t$; it is the requirement that the
parametrization be a \textbf{homeomorphism} onto its image. Here $\gamma^{-1}$ is discontinuous
at $(0,0)$: points of the image arbitrarily close to the origin have parameters near $0$ and near
$2\pi$, far apart in $I$. That the resulting set is not a submanifold can then be confirmed by
the component count of \cref{ex:cross_not_submanifold}. Deleting the crossing point leaves
four pieces, while deleting a point from an interval leaves two.
\end{ainote}

\section{A second quiz: the integration half}
\label{sec:repetition_quiz_integration}

Corsin's quiz stops where integration begins, so on its own it tests
\crefrange{ch:prerequisites}{ch:submanifolds} and leaves
\crefrange{ch:jordan_measure}{ch:stokes} untested. The seven questions below close that gap, in
the same format and at the same level: each is answerable from a definition or a theorem
statement, and none needs a long computation. They are best attempted with the book shut.

\begin{ainote}
% Generator: Claude Opus 5 (High)
These seven questions and their answers were generated by Claude Opus 5 for these notes. Unlike
\cref{sec:repetition_quiz}, they are not from the course, and no claim is made that they resemble
what an examiner would ask. What they do reproduce is the shape of the mid-semester quiz: each one
is built around a hypothesis that is easy to forget, and the answer says which hypothesis it is.
\end{ainote}

\begin{question}
\label{q:quiz_q8}
Every bounded \emph{open} subset of $\mathbb{R}^n$ is Jordan measurable. True or false?
\end{question}

\begin{answer}
\textbf{False.} By \cref{thm:jordan_measurability_boundary_criterion}, Jordan measurability is a
condition on the boundary, not on the interior, and an open set can have a boundary of positive
outer measure. \Cref{ex:non_jordan_measurable_open_set} is the standard witness: a union of
shrinking intervals around an enumeration of $\mathbb{Q}\cap[0,1]$, open and bounded, with
$\mu_{\mathrm{in}} \leq \tfrac12$ and $\mu_{\mathrm{out}} = 1$.
\end{answer}

\begin{ainote}
The instinct to be corrected is that \qt{nice} should mean open or closed. It does not: a closed
set can fail just as badly, since the complement construction is available in both directions,
and $\overline U = [0,1]$ in the example above is Jordan measurable while $U$ is not. What Jordan
measurability actually asks is that the boundary be thin, and \emph{thin} is a size condition
that topology alone never supplies.
\end{ainote}

\begin{question}
\label{q:quiz_q9}
Let $\Phi : U \to V$ be a $C^1$-diffeomorphism, let $A \subseteq U$ be Jordan measurable with
$\overline A \subseteq U$, and let $f : \Phi(A) \to \mathbb{R}$ be continuous. Which of the
following hold?
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item $\displaystyle\int_{\Phi(A)} f(y)\,dy
    = \int_A f\big(\Phi(x)\big)\,\big\lvert\det\Jac\Phi(x)\big\rvert\,dx$.
  \item $\displaystyle\int_{\Phi(A)} f(y)\,dy
    = \int_A f\big(\Phi(x)\big)\,\big\lvert\det\Jac\Phi(x)\big\rvert^{-1}\,dx$.
  \item Both, since
    $\big\lvert\det\Jac\Phi\big(\Phi^{-1}(y)\big)\big\rvert
      = \big\lvert\det\Jac\Phi^{-1}(y)\big\rvert$.
\end{enumerate}
\end{question}

\begin{answer}
\textbf{(a) is true; (b) and (c) are false.} Statement \textbf{(a)} is
\cref{thm:change_of_variables} with $\Phi$ moved to the other side, which is where the
determinant stops being divided and starts being multiplied. Statement \textbf{(c)} misstates
the inverse function theorem: differentiating $\Phi^{-1}\circ\Phi = \id$ gives
\[\Jac\Phi^{-1}(y) = \Big(\Jac\Phi\big(\Phi^{-1}(y)\big)\Big)^{-1},
  \qquad\text{so}\qquad
  \det\Jac\Phi^{-1}(y) = \frac{1}{\det\Jac\Phi\big(\Phi^{-1}(y)\big)}.\]
The two determinants are reciprocals, not equals, and it is exactly that reciprocal which turns
\textbf{(a)} into the form stated in \cref{thm:change_of_variables}. Since \textbf{(a)} is true
and the two right-hand sides differ whenever $\lvert\det\Jac\Phi\rvert \neq 1$, \textbf{(b)} is
false.
\end{answer}

\begin{ainote}
The reliable way to settle which side carries the determinant is a one-line dimensional check,
and it is worth doing every time rather than memorising the formula. Take $f \equiv 1$ and
$\Phi(x) := 2x$ on $\mathbb{R}^n$. The left-hand side is $\vol_n(2A) = 2^n\vol_n(A)$, and
$\lvert\det\Jac\Phi\rvert = 2^n$, so only \textbf{(a)} balances.
\end{ainote}

\begin{question}
\label{q:quiz_q10}
Let $\phi : B \to \mathbb{R}^n$ and $\psi : \widetilde B \to \mathbb{R}^n$ be two
parametrizations of the same $m$-dimensional surface patch, and let $\vol_m$ be computed from
each by \cref{def:d_volume_parametrized_submanifold}. Do the two computations agree?
\end{question}

\begin{answer}
\textbf{True, and the Gram determinant is what makes it work.} Write $\psi = \phi\circ\Psi$ for
the transition diffeomorphism $\Psi := \phi^{-1}\circ\psi$. By the chain rule
$D\psi = (D\phi\circ\Psi)\,D\Psi$, and therefore
\[D\psi\transp D\psi = D\Psi\transp\big(D\phi\transp D\phi\big)D\Psi,
  \qquad
  \sqrt{\det\big(D\psi\transp D\psi\big)}
    = \big\lvert\det D\Psi\big\rvert\,\sqrt{\det\big(D\phi\transp D\phi\big)}\circ\Psi,\]
using $\det(D\Psi\transp M\,D\Psi) = (\det D\Psi)^2\det M$ and taking the square root, which is
where the absolute value appears. That factor $\lvert\det D\Psi\rvert$ is precisely what
\cref{thm:change_of_variables} consumes when the integral over $\widetilde B$ is rewritten as an
integral over $B$, so the two values coincide.
\end{answer}

\begin{ainote}
This is the reason \cref{def:d_volume_parametrized_submanifold} is a definition of the surface at
all, rather than of the map used to describe it. Note also which ingredient did the work: the
square root of a Gram determinant is homogeneous of degree one in the parametrization, so it
transforms with $\lvert\det D\Psi\rvert$, exactly the weight that change of variables cancels.
The signed version of the same statement is the integral of an $m$-form, which transforms with
$\det D\Psi$ instead and therefore \emph{does} depend on the parametrization, through its
orientation (see \cref{def:integral_of_form_over_submanifold}).
\end{ainote}

\begin{question}
\label{q:quiz_q11}
Let $U \subseteq \mathbb{R}^3$ be open and $F \in C^1(U,\mathbb{R}^3)$ with $\curl F \equiv 0$
on $U$. Which of the following holds?
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item $F$ is conservative on $U$.
  \item $F$ is conservative on $U$ provided $U$ is simply connected.
  \item $F$ is conservative on every ball contained in $U$.
\end{enumerate}
\end{question}

\begin{answer}
\textbf{(a) is false; (b) and (c) are true.} By \cref{lem:integrability_conditions} the
vanishing of $\curl F$ is \emph{necessary} for $F$ to be conservative, and
\cref{cor:potentials_simply_connected} makes it sufficient once the domain is simply connected,
which gives \textbf{(b)}. A ball is star-shaped, so \textbf{(c)} follows from the Poincaré lemma
(\cref{thm:poincare_lemma}) and shows that the failure in \textbf{(a)} can only ever be global.

For \textbf{(a)} itself, take $U := \mathbb{R}^3\setminus\{x = y = 0\}$, the complement of the
$z$-axis, and
\[F(x,y,z) := \left(\frac{-y}{x^2+y^2},\ \frac{x}{x^2+y^2},\ 0\right).\]
Then $\curl F = 0$ on $U$, but the line integral of $F$ around the unit circle in the plane
$z = 0$ is $2\pi \neq 0$, so no potential exists by \cref{prop:work_difference_potential}.
\end{answer}

\begin{ainote}
The field above is the angle form of \cref{ex:ai_angle_form_closed} carried into
$\mathbb{R}^3$ and given a third component of zero, and its failure has the same cause: the
domain has a hole, so the polar angle it wants as a potential cannot be defined continuously all
the way round. The three parts of this question are worth reading as one statement. Vanishing
curl is a \emph{local} condition, being conservative is a \emph{global} one, and the gap between
them is measured by the topology of $U$ and by nothing else.
\end{ainote}

\begin{question}
\label{q:quiz_q12}
Gauss' theorem (\cref{thm:gauss_divergence}) requires the boundary $\partial\Omega$ to be
connected. True or false?
\end{question}

\begin{answer}
\textbf{False.} Nothing in the hypotheses mentions connectedness of the boundary, and the
theorem is routinely used on domains where it fails. Take the spherical shell
$\Omega := \{x \in \mathbb{R}^3 : 1 < \lvert x\rvert < 2\}$, whose boundary is the disjoint
union of two spheres, and the field $F(x) := x/\lvert x\rvert^3$, which is $C^1$ on a
neighbourhood of $\overline\Omega$ and has $\divg F = 0$ there. On the outer sphere the exterior
normal is $\nu = x/2$ and $\langle F,\nu\rangle = \tfrac14$, contributing
$\tfrac14 \cdot 16\pi = 4\pi$. On the inner sphere the exterior normal of $\Omega$ points
\emph{towards} the origin, so $\nu = -x$ and $\langle F,\nu\rangle = -1$, contributing
$-4\pi$. The two add to $0 = \int_\Omega\divg F$, as the theorem requires.
\end{answer}

\begin{ainote}
The step that has to be got right on a domain like this is not the theorem but the normal: on an
inner boundary component the exterior normal of $\Omega$ points into the hole. Getting it
backwards produces $8\pi$ instead of $0$, and the error is invisible in the algebra.

It is also worth comparing with \cref{ex:11.5}. The same field, integrated over the solid ball
instead of the shell, gives a non-zero boundary integral against a vanishing divergence integral,
and there the resolution is that $F$ is singular at the origin. Excising a small ball around the
singularity turns the ball into a shell, and the contradiction into the computation above.
\end{ainote}

\begin{question}
\label{q:quiz_q13}
The Möbius strip is a $2$-dimensional submanifold of $\mathbb{R}^3$. Is it orientable?
\end{question}

\begin{answer}
\textbf{No.} An orientation in the sense of \cref{def:orientable_manifold} is an atlas whose
transition maps all have positive Jacobian determinant, and for a surface in $\mathbb{R}^3$ that
is equivalent to a continuous choice of unit normal. Transporting a normal once around the core
circle of the strip returns it reversed, so no continuous choice exists.
\end{answer}

\begin{ainote}
This is the same fact used earlier for a different purpose. In \cref{sec:recognising_submanifolds}
it was the reason the Möbius strip is not $g^{-1}\{0\}$ for any single $g$ with
$\nabla g \neq 0$: such a $g$ would hand over the global normal
$\nabla g/\lvert\nabla g\rvert$ that the strip does not have. Non-orientability is therefore not
an exotic pathology bolted on at the end of the course. It is the obstruction that was already
forcing implicit descriptions to be local.
\end{ainote}

\begin{question}
\label{q:quiz_q14}
Let $M$ be a compact orientable $m$-manifold without boundary.
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Is $\int_M d\omega = 0$ for every $(m-1)$-form $\omega$?
  \item The standard area form $\sigma$ on $\mathbb{S}^2$ satisfies
    $\int_{\mathbb{S}^2}\sigma = 4\pi$. Is $\sigma$ exact?
\end{enumerate}
\end{question}

\begin{answer}
\textbf{(a) Yes.} \Cref{thm:stokes_general} gives
$\int_M d\omega = \int_{\partial M}\omega$, and $\partial M = \emptyset$, so the right-hand side
is an integral over the empty set.

\textbf{(b) No}, and part \textbf{(a)} is the proof. If $\sigma = d\omega$ for some $1$-form
$\omega$, then $\int_{\mathbb{S}^2}\sigma = \int_{\mathbb{S}^2}d\omega = 0$ by \textbf{(a)},
contradicting $\int_{\mathbb{S}^2}\sigma = 4\pi \neq 0$. Since $\sigma$ is a top-degree form on a
$2$-manifold, $d\sigma$ is a $3$-form on a $2$-manifold and vanishes automatically, so $\sigma$ is
closed. It is therefore closed and not exact.
\end{answer}

\begin{ainote}
Part \textbf{(b)} is worth holding onto, because it is the first obstruction in this document
that is detected by an \emph{integral} rather than by a computation. The angle form of
\cref{ex:ai_angle_form_closed} was closed and not exact for the same reason, one dimension down,
and there too the witness was a number: $2\pi$ around the puncture, $4\pi$ over the sphere. In
both cases the domain is not star-shaped, so \cref{thm:poincare_lemma} does not apply, and in
both cases the integral says not merely that the lemma is unavailable but that its conclusion is
genuinely false.
\end{ainote}
