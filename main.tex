% NB: deliberately NOT \newtheorem{theorem}{Theorem}[section].
% Each content file hard-codes its own \thetheorem prefix so that the printed
% numbers stay synchronised with Prof. Biran's handwritten notes, where one
% source part (e.g. `25.isometries.v03') may be typeset here as several
% \section's. An automatic per-section reset therefore restarted the counter in
% the middle of a source part and printed e.g. "25.a.1" three times. The counter
% now resets per chapter only; a file that opens a new source part resets it
% explicitly with \setcounter{theorem}{0}.
\newtheorem{theorem}{Theorem}

% The decoupled theorem display format: Chapter.letter.Number
% Robust against chapters that jump straight to \subsection (i.e. no \section at
% all): in that case the section letter is simply omitted -> "Chapter.Number".
\renewcommand{\thetheorem}{%
  \thechapter.\ifnum\value{section}>0 \alph{section}.\fi\arabic{theorem}}

% hyperref builds its anchor names from \theH<counter>, NOT from \the<counter>.
% Its default for a counter reset by `section' omits the chapter, so e.g. the
% first theorem of every section of every chapter would claim the anchor
% `theorem.<n>.1' -- the duplicates get dropped and \cref links silently jump to
% the wrong environment. Mirroring the printed number keeps the anchors unique.
% (\AtBeginDocument: hyperref installs its own defaults late, so a plain
% \renewcommand here would simply be overwritten.)
\makeatletter
\AtBeginDocument{%
  \renewcommand{\theHtheorem}{\thechapter.\alph{section}.\arabic{theorem}}%
  \renewcommand{\theHequation}{\thechapter.\alph{section}.\arabic{equation}}%
  % The aliased counters below need the same treatment, and must point at the
  % *same* anchor as \theHtheorem -- they are one counter under many names.
  \@namedef{theHlemma}{\theHtheorem}%
  \@namedef{theHcorollary}{\theHtheorem}%
  \@namedef{theHdefinition}{\theHtheorem}%
  \@namedef{theHproposition}{\theHtheorem}%
  \@namedef{theHexercise}{\theHtheorem}%
  % Sonnet 5 (Medium): every formerly-unnumbered environment below is now
  % aliased to the theorem counter too (see "NUMBERED ENVIRONMENTS, continued"),
  % so it needs the same anchor treatment as lemma/corollary/... above.
  \@namedef{theHclaim}{\theHtheorem}%
  \@namedef{theHnotation}{\theHtheorem}%
  \@namedef{theHremark}{\theHtheorem}%
  \@namedef{theHexample}{\theHtheorem}%
  \@namedef{theHsummary}{\theHtheorem}%
  \@namedef{theHwarmup}{\theHtheorem}%
  \@namedef{theHquestion}{\theHtheorem}%
  \@namedef{theHanswer}{\theHtheorem}%
  \@namedef{theHimportantremark}{\theHtheorem}%
  \@namedef{theHgoals}{\theHtheorem}%
  \@namedef{theHconclusion}{\theHtheorem}%
  % (no theHainote: ainote is unnumbered, see its \newtheorem* below)
  \@namedef{theHaiexample}{\theHtheorem}%
  \@namedef{theHaiexercise}{\theHtheorem}%
}
\makeatother

% --- NUMBERED ENVIRONMENTS ---
% NB: deliberately NOT \newtheorem{lemma}[theorem]{Lemma}.
% That form makes `lemma' share the *counter named* `theorem', and cleveref
% derives a reference's type from the counter name -- so every \cref to a
% lemma, corollary, definition or proposition printed "Theorem". Only 52 of the
% ~290 numbered environments in this document are actually theorems, so almost
% every cross-reference was mislabelled.
%
% aliascnt gives each environment a counter of its own name that is merely an
% alias of `theorem'. The shared, continuous numbering is preserved exactly
% (\thelemma expands to \thetheorem, dynamically, so the per-file
% \renewcommand{\thetheorem} prefixes still apply), but cleveref can now tell
% the environments apart.
\newaliascnt{lemma}{theorem}
\newtheorem{lemma}[lemma]{Lemma}
\aliascntresetthe{lemma}

\newaliascnt{corollary}{theorem}
\newtheorem{corollary}[corollary]{Corollary}
\aliascntresetthe{corollary}

\newaliascnt{definition}{theorem}
\newtheorem{definition}[definition]{Definition}
\aliascntresetthe{definition}

\newaliascnt{proposition}{theorem}
\newtheorem{proposition}[proposition]{Proposition}
\aliascntresetthe{proposition}

\newaliascnt{exercise}{theorem}
\newtheorem{exercise}[exercise]{Exercise}
\aliascntresetthe{exercise}

% --- NUMBERED ENVIRONMENTS, continued ---
% Sonnet 5 (Medium): this project wants every environment numbered, so that
% cleveref/\cref can identify and cross-reference every instance -- an
% environment created with \newtheorem* has NO counter at all, so a \label
% placed inside one is silently attributed to whatever ambient counter (e.g.
% the enclosing subsection) was last stepped, producing wrong \cref output
% (e.g. "Section 2.d.4" instead of "AI-Exercise 2.d.13"). Every environment
% below therefore follows the same \newaliascnt pattern as lemma/corollary/
% definition/proposition/exercise above, so it shares the single running
% Chapter.Section.N sequence and cleveref can tell it apart by name. The

% ==========================================
% TRANSPOSE  --  one switch for the whole document
% ==========================================
% Usage: A\transp, (PAQ)\transp, \overline{v}\transp.
%
% This text uses `T' constantly for linear maps ($T$, $T_A$, $[T]$), so an
% italic $A^T$ reads as an index or a map name. An upright sans-serif T cannot
% be confused with it and survives being shrunk to superscript size, which is
% why it is preferred here over a bold T (bold at 70% mostly reads as smudged,
% and bold+colour is already reserved for \textbf / \emph in this document).
%
% To switch the whole document to a bold transpose, change \mathsf to \mathbf
% on the next line -- nothing else needs touching.
\newcommand{\transposesymbol}{\mathsf{T}}
\newcommand{\transp}{^{\transposesymbol}}
