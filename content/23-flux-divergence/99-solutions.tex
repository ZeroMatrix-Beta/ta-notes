\begin{exercisesolution}[Solution of \cref{ex:radial_vector_fields_flux}]
\label{sol:radial_vector_fields_flux}
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Writing $V_\alpha^i(x) = \lvert x\rvert^\alpha x_i$,
    \[
    \partial_i\big(\lvert x\rvert^\alpha x_i\big) = \alpha\lvert x\rvert^{\alpha-2}x_i^2
      + \lvert x\rvert^\alpha \qquad \text{(no sum),}
    \]
    so, summing over $i=1,\ldots,n$,
    \[
    \divg V_\alpha(x) = \sum_{i=1}^n \Big(\alpha\lvert x\rvert^{\alpha-2}x_i^2 + \lvert
      x\rvert^\alpha\Big) = \alpha\lvert x\rvert^{\alpha-2}\lvert x\rvert^2
      + n\lvert x\rvert^\alpha = (\alpha+n)\lvert x\rvert^\alpha.
    \]
    Since $\lvert x\rvert^\alpha \neq 0$ for $x\neq0$, this vanishes identically exactly when
    $\alpha = -n$.
  \item For $\alpha=0$, $V_0(x)=x$ and, by part \textbf{(a)}, $\divg V_0 \equiv n$. Applying
    \cref{thm:gauss_divergence} on $B_r$,
    \[\int_{B_r} \divg V_0\,dx = n\,\vol_n(B_r).\]
    On $\partial B_r$ the outward unit normal is $\nu(x) = x/r$, so $V_0(x)\cdot\nu(x) = x\cdot
    x/r = \lvert x\rvert^2/r = r$ (using $\lvert x\rvert=r$ on $\partial B_r$), giving
    \[\int_{\partial B_r} V_0\cdot\nu\,d\vol_{n-1} = r\,\vol_{n-1}(\partial B_r).\]
    Equating the two expressions of $\int_{B_r}\divg V_0\,dx = \int_{\partial B_r}V_0\cdot\nu\,
    d\vol_{n-1}$ gives $r\,\vol_{n-1}(\partial B_r) = n\,\vol_n(B_r)$.
  \item By part \textbf{(a)}, $\divg V_{-n}(x) = 0$ for every $x\neq0$, so $\int_{B_1}\divg
    V_{-n}(x)\,dx = 0$ trivially (the integrand vanishes on $B_1\setminus\{0\}$, a set of full
    measure in $B_1$). On $\partial B_1$, $\nu(x)=x$ and $\lvert x\rvert=1$, so $V_{-n}(x) =
    \lvert x\rvert^{-n}x = x$, giving
    \[\int_{\partial B_1} V_{-n}\cdot\nu\,d\vol_{n-1} = \int_{\partial B_1} x\cdot x\,d\vol_{n-1}
      = \int_{\partial B_1} 1\,d\vol_{n-1} = \vol_{n-1}(\partial B_1) > 0.\]
    This does not contradict \cref{thm:gauss_divergence}: the theorem requires $F$ to be $C^1$ on
    all of $\overline{B_1}$, but $V_{-n}(x) = \lvert x\rvert^{-n}x$ is not even defined, let
    alone continuously differentiable, at $x=0 \in \overline{B_1}$. Its hypotheses simply do not
    hold on this domain.
\end{enumerate}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:ai_flux_cubic_sphere}]
\label{sol:ai_flux_cubic_sphere}
% Generator: Claude Opus 5 (High)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item On the unit sphere the exterior unit normal is $\nu(x,y,z) = (x,y,z)$, so the
    integrand of \cref{def:flux} is $\langle F,\nu\rangle = x^4+y^4+z^4$. In the spherical
    parametrization that becomes
    $\sin^4\theta\cos^4\varphi + \sin^4\theta\sin^4\varphi + \cos^4\theta$, multiplied by the
    surface element $\sin\theta$: a fourth-power trigonometric integral in two variables, where
    the divergence theorem offers a radial integral in one.
  \item Writing $r$ for the distance to the origin,
    $\divg F = 3x^2+3y^2+3z^2 = 3r^2$, which is radial.
  \item Because the integrand is radial, \cref{rem:polar_as_shells} applies with $n = 3$ and
    $C_3 = 4\pi$:
    \[\int_{\partial\Omega}\langle F,\nu\rangle\,d\vol_2 = \int_{B_1(0)} 3r^2\,d\vol_3
      = 4\pi\int_0^1 3r^2\cdot r^2\,dr = 12\pi\int_0^1 r^4\,dr = \frac{12\pi}{5}.\]
\end{enumerate}

\begin{remark}[The direct route, for once, is also available]
\label{rem:cubic_flux_direct_check}
The surface integral abandoned in part \textbf{(a)} is not actually intractable, and working it
out is a
worthwhile check. By the symmetry of the sphere under permuting the coordinates, the three terms
of $x^4+y^4+z^4$ contribute equally, so
\[\int_{\mathbb{S}^2}\big(x^4+y^4+z^4\big)\,d\vol_2 = 3\int_{\mathbb{S}^2} z^4\,d\vol_2
  = 3\int_0^{2\pi}\!\!\int_0^\pi \cos^4\theta\,\sin\theta\,d\theta\,d\varphi
  = 6\pi\left[-\frac{\cos^5\theta}{5}\right]_0^\pi = \frac{12\pi}{5},\]
in agreement. Note that what made this bearable was a symmetry argument, applied \emph{before}
any trigonometry: expanding all three fourth powers first is the version that is unpleasant.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:ai_capping_a_paraboloid}]
\label{sol:ai_capping_a_paraboloid}
% Generator: Claude Opus 5 (High)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Take $\Omega := \{(x,y,z) : x^2+y^2 < 1,\ 0 < z < 1-x^2-y^2\}$, the solid region
    between the paraboloid and the plane $z = 0$. Its boundary is $S$ together with the flat disk
    \[D := \{(x,y,0) : x^2+y^2 \leq 1\},\]
    whose exterior unit normal is $\nu \equiv -e_3$, since $\Omega$ lies above it. On $S$ the
    exterior normal is the upward-pointing one, which is the orientation the exercise already
    fixes, so no sign has to be corrected later.
  \item Here $\divg F = 1+1+1 = 3$, so the flux out of $\partial\Omega$ is three times a
    volume. In cylindrical coordinates,
    \[\vol_3(\Omega) = \int_0^{2\pi}\!\!\int_0^1\!\!\int_0^{1-r^2} r\,dz\,dr\,d\varphi
      = 2\pi\int_0^1 \big(r - r^3\big)\,dr = 2\pi\left(\frac12 - \frac14\right) = \frac{\pi}{2},\]
    and therefore
    \[\int_{\partial\Omega}\langle F,\nu\rangle\,d\vol_2 = 3\,\vol_3(\Omega) = \frac{3\pi}{2}.\]
  \item Along $D$ the third coordinate is $z = 0$, so
    $\langle F,\nu\rangle = \langle (x,y,0),\,-e_3\rangle = -z = 0$ at every point: the field is
    tangent to the cap, and contributes nothing. Splitting the boundary integral,
    \[\int_S \langle F,\nu\rangle\,d\vol_2
      = \int_{\partial\Omega}\langle F,\nu\rangle\,d\vol_2 - \int_D \langle F,\nu\rangle\,d\vol_2
      = \frac{3\pi}{2} - 0 = \frac{3\pi}{2}.\]
\end{enumerate}

\begin{remark}[The cap vanished because of this field, not because it is a cap]
\label{rem:cap_contribution_is_not_generally_zero}
It is tempting to read part \textbf{(c)} as licence to ignore the lid. Change $F$ to
$\widetilde F(x,y,z) := (x,\,y,\,z+1)$ and nothing about the first two parts moves, since
$\divg\widetilde F = 3$ as before and the closed flux is still $3\pi/2$. But now
$\langle \widetilde F,\nu\rangle = -1$ on $D$, so the cap carries a flux of $-\pi$, and the flux
through the paraboloid is $3\pi/2 + \pi = 5\pi/2$ instead. The subtraction in part \textbf{(c)}
is doing
real work; it merely had nothing to subtract.

\Cref{ex:vector_field_flux_hemisphere} is the same technique on a hemisphere, and the two are worth reading as a pair:
there the cap contributes $\pi/2$ and must be subtracted, here it contributes nothing. The
method is identical and only the arithmetic differs, which is the point.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:examHS24_KP1_part2_flux}]
\label{sol:examHS24_KP1_part2_flux}
% Generator: Gemini 3.1 Pro (High)
Both fields are $C^1$ on all of $\mathbb{R}^3$, so the divergence theorem
(\cref{thm:gauss_divergence}) converts the flux into a volume integral:
\[\int_{\partial U} \langle B, \nu\rangle \, d\vol_2 = \int_U \divg B \, d\vol_3 .\]
The summand $\curl A$ contributes nothing, since the divergence of a curl vanishes identically.
Only the first summand has to be differentiated:
\begin{align*}
  \divg B &= \partial_x\big(x - \sinh y\big) + \partial_y\big(y + \cosh x\big)
             + \partial_z\big(z + e^{-\pi(x^2+y^2)}\big) \\
          &= 1 + 1 + 1 = 3 .
\end{align*}
The flux is therefore $3\vol_3(U)$, and the solid ellipsoid with semi-axes $a$, $b$, $c$ has volume
$\tfrac43\pi abc$. Consequently, the outer flow of $B$ across $\partial U$ is
\[3 \cdot \tfrac43\pi abc = 4\pi abc .\]
\begin{remark}
Every intimidating term in $B$ is decoration. Each of $\sinh y$, $\cosh x$ and
$e^{-\pi(x^2+y^2)}$ depends only on the variables the matching partial derivative does not
differentiate, so all three vanish, and $\curl A$ is invisible to the divergence whatever $A$ is.
What survives is the radial field $(x,y,z)$, whose divergence is $3$.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{ex:gauss_law_ellipsoid_flux}]
\label{sol:gauss_law_ellipsoid_flux}
% Generator: Gemini 3.6 Flash (High)
% Correction: Claude Opus 5 (High) --- \operatorname{div}/\operatorname{vol} -> \divg/\vol, and the
% triple and double integral signs collapsed to \int, matching thm:gauss_divergence.
% Correction: Claude Opus 5 (High) --- a second, later solution for this same exercise was appended
% to the end of this file by a Gemini 3.1 Pro pass and has been deleted; this is the original.
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Write $r := \lvert x\rvert = \sqrt{x^2+y^2+z^2}$, so that $E(x) = x/r^3$. Since
    $\partial_i r = x_i/r$, the product rule gives
\[\divg E = \sum_{i=1}^3 \partial_i \left(x_i r^{-3}\right) = \sum_{i=1}^3 \left(r^{-3} - 3 x_i r^{-4} \frac{x_i}{r}\right) = 3r^{-3} - 3r^{-5} \lvert x\rvert^2 = 0 \qquad (r > 0).\]

  \item Choose $r_0 > 0$ small enough that $\overline{B_{r_0}(0)} \subseteq U$, and set
    $\Omega := U \setminus \overline{B_{r_0}(0)}$, so that $\partial \Omega = \partial U \cup \partial B_{r_0}(0)$.
    The point $0$ has been removed, so $E$ is smooth on $\overline{\Omega}$ and part \textbf{(a)}
    applies throughout $\Omega$. The divergence theorem (\cref{thm:gauss_divergence}) therefore gives
\[0 = \int_\Omega \divg E \, d\vol_3 = \int_{\partial U} \langle E,\nu\rangle \, d\vol_2 + \int_{\partial B_{r_0}(0)} \langle E,\nu_{\mathrm{in}}\rangle \, d\vol_2,\]
    where on the inner sphere the outward normal \emph{of $\Omega$} points towards the origin,
    $\nu_{\mathrm{in}} = -x/r_0$. Consequently
\[\int_{\partial U} \langle E,\nu\rangle \, d\vol_2 = \int_{\partial B_{r_0}(0)} \left\langle E, \frac{x}{r_0}\right\rangle d\vol_2 .\]
    On $\partial B_{r_0}(0)$ we have $E(x) = x/r_0^3$, so the integrand is the constant
    $\lvert x\rvert^2/r_0^4 = r_0^2/r_0^4 = r_0^{-2}$, and
\[\int_{\partial U} \langle E,\nu\rangle \, d\vol_2 = \frac{1}{r_0^2}\,\vol_2(\partial B_{r_0}(0)) = \frac{4\pi r_0^2}{r_0^2} = 4\pi.\]
\end{enumerate}
\begin{remark}
The answer does not depend on $a$, $b$, $c$ --- or, for that matter, on the shape of $U$ at all,
only on whether it encloses the origin. That is Gauss' law: the flux counts the charge inside, and
nothing else. The $r_0$ in the argument is a device that never appears in the answer, which is the
usual sign that the quantity computed is a topological one.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Solution of \cref{ex:tuning_alpha_prescribed_flux}]
\label{sol:tuning_alpha_prescribed_flux}
% Generator: Claude Opus 5 (High)
The field is polynomial, hence $C^1$, and $U$ is a smooth bounded domain, so the divergence theorem
(\cref{thm:gauss_divergence}) converts the flux into a volume integral. The divergence is a constant:
\[\divg F_\alpha = \partial_x\big(3y^2 + \alpha x\big) + \partial_y\big(2x - y\big) = \alpha - 1 .\]
Therefore
\[\int_{\partial U} \langle F_\alpha, \nu\rangle \, d\vol_1 = \int_U \divg F_\alpha \, d\vol_2
  = (\alpha - 1)\,\vol_2(U) = \alpha - 1,\]
using $\vol_2(U) = 1$. Setting this equal to $1$ gives $\boxed{\alpha = 2}$.
\begin{remark}
The terms $3y^2$ and $2x$ are decoration: they contribute nothing to the divergence, so nothing to
the flux, however violently they stir the field around inside $U$. Only the parts of $F_\alpha$ that
create or destroy field lines --- here $\alpha x$ and $-y$ --- can be seen from the boundary. Note
also that the answer is independent of the shape of $U$ and uses only its area.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{ex:radial_weight_flux_independent}]
\label{sol:radial_weight_flux_independent}
% Generator: Claude Opus 5 (High)
Write $w_n(x) := \big(\lvert x\rvert^2\big)^n = (x_1^2+x_2^2+x_3^2)^n$. This is a polynomial, so
$w_n v$ is a $C^1$ vector field on $\mathbb{R}^3$ and the divergence theorem
(\cref{thm:gauss_divergence}) applies on the unit ball:
\[I_n = \int_{B_1(0)} \divg\big(w_n v\big) \, d\vol_3
      = \int_{\partial B_1(0)} \big\langle w_n v, \nu\big\rangle \, d\vol_2 .\]
The whole point is now visible: the integration has been moved to the unit sphere, and there
$\lvert x\rvert = 1$, so $w_n(x) = 1^n = 1$ for \emph{every} $n$. Hence
\[I_n = \int_{\partial B_1(0)} \langle v, \nu\rangle \, d\vol_2 ,\]
an expression in which $n$ no longer appears. Since this holds for every $n \in \mathbb{Z}_{\ge0}$,
we get $I_n = I_m$ for all $n, m$, all of them equal to the flux of $v$ itself through the unit
sphere.
\begin{ainote}
It is worth seeing how much work is avoided. Expanding $\divg(w_n v) = w_n \divg v + \langle \nabla
w_n, v\rangle$ with $\nabla w_n = 2n\lvert x\rvert^{2n-2}x$ gives a genuinely messy integrand whose
$n$-dependence is not remotely obvious, and no amount of manipulating it inside the ball will make
the cancellation appear. The divergence theorem removes the difficulty by moving to a surface on
which the offending factor is identically $1$. The radius $1$ is doing real work here: on
$\partial B_R(0)$ the same argument would give $I_n = R^{2n}\int\langle v,\nu\rangle$, which does
depend on $n$ unless $R = 1$.
\end{ainote}
\end{exercisesolution}

\begin{exercisesolution}[Solution of \cref{ex:flux_truncated_paraboloid}]
\label{sol:flux_truncated_paraboloid}
% Generator: Claude Opus 5 (High)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item $M$ is the paraboloid of revolution $z = 1+x^2+y^2$, opening upwards, with vertex at
    $(0,0,1)$ and cut off at height $z = 5$. The cut happens where $x^2+y^2 = 4$, so the rim is the
    circle of radius $2$ at height $5$.

%   vertex  y=0, z=1  -> canvas (0, 0.7)

% points above, so the parabola meets the rim ellipse exactly at its left and right extremes.

  \item The field $E$ has
    \[\divg E = \partial_x(xz) + \partial_y(2y-yz) + \partial_z(x^2+y^2) = z + (2-z) + 0 = 2 .\]
    The boundary of $U$ consists of $M$ together with the flat lid
    $D := \{(x,y,5) \mid x^2+y^2 < 4\}$, and the divergence theorem in its piecewise form
    (\cref{thm:gauss_divergence_piecewise}) gives
    \[\int_M \langle E,\nu\rangle\,d\vol_2 + \int_D \langle E,\nu\rangle\,d\vol_2
      = \int_U \divg E\,d\vol_3 = 2\vol_3(U).\]
    In cylindrical coordinates the region lies over the disk of radius $2$, between the paraboloid
    and the plane $z=5$, so
    \[\vol_3(U) = \int_0^{2\pi}\!\!\int_0^2 \big(5 - (1+r^2)\big)\,r\,dr\,d\theta
      = 2\pi\int_0^2 (4-r^2)\,r\,dr = 2\pi\Big[2r^2 - \tfrac{r^4}{4}\Big]_0^2
      = 2\pi(8-4) = 8\pi,\]
    and therefore the total outer flux across $\partial U$ is $16\pi$.

    On $D$ the exterior normal is $\nu = e_3$, so the integrand is the third component of $E$,
    namely $x^2+y^2$, evaluated at $z=5$:
    \[\int_D \langle E,\nu\rangle\,d\vol_2 = \int_0^{2\pi}\!\!\int_0^2 r^2\cdot r\,dr\,d\theta
      = 2\pi\Big[\tfrac{r^4}{4}\Big]_0^2 = 8\pi .\]
    Subtracting,
    \[\boxed{\ \int_M \langle E,\nu\rangle\,d\vol_2 = 16\pi - 8\pi = 8\pi.\ }\]

  \item First compute the curl, using the formula of \cref{def:rotation}:
    \[\curl A = \big(\partial_yA_3 - \partial_zA_2,\ \partial_zA_1 - \partial_xA_3,\
      \partial_xA_2 - \partial_yA_1\big)\transp
      = \big(-xe^z,\ -ye^z,\ 2e^z\big)\transp,\]
    since $\partial_yA_3 = 0$ and $\partial_zA_2 = xe^z$; $\partial_zA_1 = -ye^z$ and
    $\partial_xA_3 = 0$; and $\partial_xA_2 = e^z$ against $\partial_yA_1 = -e^z$.

    A curl is divergence-free, and here that is visible directly:
    $\divg(\curl A) = -e^z - e^z + 2e^z = 0$. Consequently the divergence theorem gives
    \emph{zero} total flux across $\partial U$, so the flux through $M$ is the negative of the flux
    through $D$. On $D$ the normal is again $e_3$ and $z = 5$ throughout, so the integrand is the
    constant $2e^5$ and
    \[\int_D \langle \curl A,\nu\rangle\,d\vol_2 = 2e^5 \cdot \vol_2(D) = 2e^5\cdot\pi\cdot 2^2
      = 8\pi e^5 .\]
    Therefore
    \[\boxed{\ \int_M \langle \curl A,\nu\rangle\,d\vol_2 = -8\pi e^5.\ }\]
\end{enumerate}
\begin{ainote}
The version of the divergence theorem this needs is \cref{thm:gauss_divergence_piecewise}, not
\cref{thm:gauss_divergence}: the region $U$ has an edge along the rim circle
$\{x^2+y^2=4,\ z=5\}$, where the paraboloid meets the lid at a nonzero angle and the exterior
normal jumps, so it is not a $C^1$ domain. That is precisely what the exam's hint licenses, by
allowing the divergence theorem on a piecewise smooth domain without proof. Nothing in the
computation is affected, because the rim carries no $\vol_2$.
\end{ainote}
\begin{remark}[Checking part \textbf{(c)} against Stokes]
Part \textbf{(c)} never used the shape of $M$, only that $M$ and $D$ share a boundary, which is a
sign that Stokes' theorem should give the same answer directly. The rim is
$c(t) := (2\cos t,\ 2\sin t,\ 5)$, and on it
\[A(c(t)) = \big(-2\sin t\,e^5,\ 2\cos t\,e^5,\ e^5\big), \qquad
  \dot c(t) = (-2\sin t,\ 2\cos t,\ 0),\]
so $\langle A(c),\dot c\rangle = 4e^5(\sin^2 t + \cos^2 t) = 4e^5$ and
$\oint_c A\cdot dc = 8\pi e^5$. That is the flux of $\curl A$ through any surface spanning the rim
whose orientation induces \emph{this} traversal, which for the lid means $\nu = e_3$: it matches
the $8\pi e^5$ computed above. The exterior normal on $M$ points downwards and outwards, inducing
the opposite traversal, and the sign flips to $-8\pi e^5$.

The same shortcut does not apply to part \textbf{(b)}, since $E$ is not a curl, and one really does
have to compute a volume there.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[\cref{ex:knoerrer_divergence_free_compact_support}]
\label{sol:knoerrer_divergence_free_compact_support}
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item The product rule for the divergence of a scalar multiple of a vector field reads
    \[\divg(fK) = \langle \nabla f, K\rangle + f \divg K.\]
    The hypothesis $\divg K = 0$ kills the second term, so $\langle\nabla f,K\rangle = \divg(fK)$
    everywhere. The field $fK$ is smooth, so \cref{thm:gauss_divergence} applies on the unit ball
    and gives
    \[\int_B \langle\nabla f, K\rangle \,d\vol_3 = \int_B \divg(fK)\,d\vol_3
      = \int_{\partial B} f\,\langle K,\nu\rangle \,d\vol_2 .\]
    It remains to see that the boundary integral vanishes. $K$ vanishes on
    $\mathbb{R}^3\setminus B$, and every point of $\partial B$ is a limit of points outside $B$,
    for instance $x = \lim_{t \downarrow 1} tx$ for $\lvert x\rvert = 1$. Since $K$ is continuous,
    $K(x) = \lim_{t\downarrow1}K(tx) = 0$ for every $x \in \partial B$. The integrand is therefore
    identically zero on $\partial B$, and
    $\int_B \langle\nabla f,K\rangle\,d\vol_3 = 0$.

  \item Apply \textbf{(a)} to the smooth function $f(x) := x_1$, whose gradient is the constant
    field $\nabla f = e_1$. Then $\langle\nabla f,K\rangle = \langle e_1,K\rangle = K_1$, so
    \[\int_B K_1 \,d\vol_3 = \int_B \langle\nabla f,K\rangle\,d\vol_3 = 0.\qedhere\]
\end{enumerate}
\begin{remark}[The hypotheses are used one each, and the second part is free]
Part \textbf{(a)} spends $\divg K = 0$ on turning the integrand into a divergence, and spends the
vanishing of $K$ outside $B$ on killing the boundary term. Neither hypothesis does anything else,
and dropping either breaks the statement: without $\divg K = 0$ a term $\int_B f \divg K$ survives,
and without the support condition the flux of $fK$ across the sphere need not vanish.

Part \textbf{(b)} is the reason to prove \textbf{(a)} in the generality of an arbitrary $f$. Taking
$f(x) = x_1$ is a device: the coordinate function is the cheapest smooth function whose gradient is
a fixed basis vector, so a statement about all $\langle\nabla f, K\rangle$ specialises to one about
a single component of $K$. Running the same trick on $x_2$ and $x_3$ gives $\int_B K = 0$ as a
vector identity, which is the real content: \textbf{a divergence-free field with compact support
has zero mean.} That is a standard fact about incompressible flows and is worth remembering in
that form.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{ex:green_first_identity_harmonic}]
\label{sol:green_first_identity_harmonic}
% Generator: Claude Opus 5 (High) --- the statement is the examiner's; the proof and the closing
% remark are written for these notes. Probeprfg1 ships no solution key.
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Apply \cref{thm:gauss_divergence} to the vector field $F := u\nabla u$, which is $C^1$ on
    $U$ because $u$ is $C^2$. Its divergence is computed by the product rule,
    \[\divg(u\nabla u) = \langle\nabla u,\nabla u\rangle + u\,\divg(\nabla u)
      = \lvert\nabla u\rvert^2 + u\,\Delta u = \lvert\nabla u\rvert^2 ,\]
    the last step because $u$ is harmonic. On the boundary,
    $\langle F,\nu\rangle = u\,\langle\nabla u,\nu\rangle = u\,\partial_\nu u$. The divergence
    theorem therefore reads
    \[\int_B \lvert\nabla u\rvert^2 \,d\vol_3 = \int_B \divg F\,d\vol_3
      = \int_{\partial B}\langle F,\nu\rangle\,d\vol_2
      = \int_{\partial B} u\,\partial_\nu u \,d\vol_2 .\]
  \item Either hypothesis makes the integrand $u\,\partial_\nu u$ vanish identically on
    $\partial B$, so part \textbf{(a)} gives
    \[\int_B \lvert\nabla u\rvert^2 \,d\vol_3 = 0 .\]
    The integrand is continuous and non-negative, so it vanishes identically on $B$: if
    $\lvert\nabla u(p)\rvert^2 = c > 0$ at some $p$ in the interior, continuity would give a ball
    $B_r(p) \subseteq B$ on which the integrand exceeds $c/2$, forcing the integral to be at
    least $\tfrac{c}{2}\vol_3(B_r(p)) > 0$; and the boundary contributes nothing to a volume
    integral. Hence $\nabla u \equiv 0$ on $B$.

    A function with vanishing gradient on a connected set is constant there. Concretely, join two
    points $p,q$ of $B$ by a $C^1$ path $\gamma$ inside $B$ and differentiate along it
    (\cref{sec:chain_rule_along_path}):
    \[u(q) - u(p) = \int_0^1 \frac{d}{dt}u(\gamma(t))\,dt
      = \int_0^1 \big\langle \nabla u(\gamma(t)),\ \gamma'(t)\big\rangle\,dt = 0 .\]
\end{enumerate}

\begin{remark}[What this buys: uniqueness for the two boundary value problems]
Read the two hypotheses of part \textbf{(b)} as data prescribed on $\partial B$ and the statement
becomes a uniqueness theorem. Suppose $u_1$ and $u_2$ are harmonic on $U$ and agree on
$\partial B$. Their difference $u := u_1 - u_2$ is harmonic with $u|_{\partial B} = 0$, so
\textbf{(b)} makes it constant, and the constant is $0$ because it vanishes on the boundary:
\emph{a harmonic function is determined on $B$ by its boundary values}. That is uniqueness for
the Dirichlet problem. The second hypothesis gives the same for the Neumann problem, where the
normal derivative is prescribed instead, except that there the conclusion is weaker by exactly
one constant, and necessarily so, since adding a constant changes neither $\Delta u$ nor
$\partial_\nu u$.

The mechanism is worth naming. Part \textbf{(a)} is \newterm{Green's first identity}, and it is
integration by parts: the product rule for $\divg(u\nabla u)$, which is
\cref{ex:differential_identities_true_false}\textbf{(d)}, trades a derivative from one factor to
the other at the cost of a boundary term. The harmonicity is what kills the term that would
otherwise survive, and the sign is what makes the argument work at all --- $\lvert\nabla
u\rvert^2$ has one sign, so a vanishing integral forces a vanishing integrand, which no
indefinite quantity would allow.
\end{remark}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{ex:flux_through_nested_spheres}]
\label{sol:flux_through_nested_spheres}
% Generator: Claude Opus 5 (High) --- the four statements are the examiner's; the working is
% written for these notes. Verdicts checked afterwards against Probeprfg3_Lsg.pdf.
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item \textbf{False}, and the counterexample is the one field this situation was invented for.
    Take the inverse-square field
    \[f(x) := \frac{x}{\lvert x\rvert^3}, \qquad x \in U .\]
    It is divergence free on $U$:
    \[\divg f = \sum_i \partial_i\big(x_i\lvert x\rvert^{-3}\big)
      = \sum_i\Big(\lvert x\rvert^{-3} - 3x_i^2\lvert x\rvert^{-5}\Big)
      = 3\lvert x\rvert^{-3} - 3\lvert x\rvert^{-3} = 0 .\]
    But on $\partial B_r(0)$ the outward normal is $\nu = x/r$, so
    $\langle f,\nu\rangle = \lvert x\rvert^2/r^4 = 1/r^2$, a constant, and
    \[I_r = \frac{1}{r^2}\vol_2\big(\partial B_r(0)\big) = \frac{4\pi r^2}{r^2} = 4\pi \neq 0 .\]
  \item \textbf{True}, and note that the field above already illustrates it: $I_r = 4\pi$ for
    every $r$. In general, fix $0 < r_1 < r_2$ and integrate over the shell
    $A := \{x \mid r_1 \leq \lvert x\rvert \leq r_2\}$, which is a compact smoothly bounded region
    contained in $U$ --- this is the step the puncture at the origin does not obstruct, and it is
    why the hypothesis is enough. Its boundary is the two spheres, and the outward normal of $A$
    on the inner one is $-\nu$, pointing towards the origin. So \cref{thm:gauss_divergence} gives
    \[0 = \int_A \divg f\,d\vol_3 = I_{r_2} - I_{r_1} .\]
  \item \textbf{False.} Take $f(x) := x$, which has $\curl f = 0$ since it is the gradient of
    $\lvert x\rvert^2/2$. Here $\langle f,\nu\rangle = \langle x, x/r\rangle = r$ on
    $\partial B_r(0)$, so
    \[I_r = r\cdot 4\pi r^2 = 4\pi r^3 ,\]
    which depends on $r$ as much as anything could. Being curl free constrains circulation, not
    flux, and the two conditions answer different questions.
  \item \textbf{True.} By \cref{prop:curl_grad_div_curl_zero} the field $\curl g$ is divergence
    free, so part \textbf{(b)} already makes $I_r$ independent of $r$ --- but that alone does not
    give the value $0$, as part \textbf{(a)} shows, and it cannot be obtained by letting
    $r \to 0$, since $g$ need not extend across the origin. What settles it is that a sphere has
    no boundary. Cut $\partial B_r(0)$ along its equator $C$ into the upper and lower caps
    $S^{\pm}$. Applying \cref{thm:stokes_classical} to each,
    \[\int_{S^+}\langle\curl g,\nu\rangle\,d\vol_2 = \oint_{C} g\cdot d\gamma,
      \qquad
      \int_{S^-}\langle\curl g,\nu\rangle\,d\vol_2 = -\oint_{C} g\cdot d\gamma,\]
    the sign differing because the outward normal orients the equator oppositely as the boundary
    of the two caps. Adding them gives $I_r = 0$.
\end{enumerate}

\begin{remark}[Divergence free is weaker than being a curl, and this is what measures the gap]
Parts \textbf{(a)}, \textbf{(b)} and \textbf{(d)} together say something sharper than any of them
says alone. On $U = \mathbb{R}^3\setminus\{0\}$ every divergence-free field has a flux
independent of $r$, every field of the form $\curl g$ has that constant equal to zero, and the
inverse-square field has it equal to $4\pi$. Therefore
\[\frac{x}{\lvert x\rvert^3} \text{ is divergence free on } U
  \quad\text{but is \emph{not} } \curl g \text{ for any } g \text{ on } U .\]
The number $4\pi$ is an obstruction, and it is detecting the hole: on a ball, or on any
star-shaped domain, no such field exists and every divergence-free field is a curl. That is the
degree-two case of the Poincar\'e lemma (\cref{thm:poincare_lemma}), whose degree-one case
\cref{ex:vortex_field_not_conservative} exhibits with the same domain punctured along a line
instead of a point.

In physics the same computation is Gauss's law, and $4\pi$ is why the charge enclosed can be read
off a flux at any radius whatever.
\end{remark}
\end{exercisesolution}
