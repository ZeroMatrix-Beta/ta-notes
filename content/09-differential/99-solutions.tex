\begin{exercisesolution}[Solution to \cref{ex:derivative_computation_exponential}]
\label{sol:derivative_computation_exponential}
% Generator: Claude Sonnet 5 (Medium)
Write $u(x,y) = x^{\sin(y)} = e^{\sin(y)\ln x}$ for $x > 0$. Differentiating with the chain rule:
\[
\partial_x u = \sin(y)\,x^{\sin(y)-1}, \qquad \partial_y u = \cos(y)\ln(x)\,x^{\sin(y)} = \cos(y)\ln(x)\,e^{\sin(y)\ln x}.
\]
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{ex:ai_differential_linearization}]
\label{sol:ai_differential_linearization}
% Generator: Gemini 3.6 Flash (Medium)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item Partial derivatives are $\frac{\partial f}{\partial x} = 2xy + y\cos(xy)$ and $\frac{\partial f}{\partial y} = x^2 + x\cos(xy)$. Evaluating at $(1,0)$ yields $\frac{\partial f}{\partial x}(1,0) = 0$ and $\frac{\partial f}{\partial y}(1,0) = 1 + 1 = 2$. Thus $\nabla f(1,0) = (0, 2)\transp$, and the total differential $Df(1,0) \colon \mathbb{R}^2 \to \mathbb{R}$ is $Df(1,0)(h_1, h_2) = 2h_2$.
  \item Since $f(1,0) = 0$, the linearization at $(1,0)$ is $L(x,y) = f(1,0) + Df(1,0)(x-1, y-0) = 0 + 0(x-1) + 2y = 2y$.
\end{enumerate}
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{ex:examHS24_TF_diff_piecewise}]
\label{sol:piecewise_differentiability_tf}
% Generator: Gemini 3.1 Pro (High)
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item \textbf{True:} For $y > 0$ and $y < 0$ the function is given by continuous formulas. On the interface $y=0$, we have $\lim_{y \to 0^+} y^2 \cos x = 0 = \lim_{y \to 0^-} 0$, which matches $f(x,0) = 0$.
  \item \textbf{True:} For $y > 0$, $\partial_y f = 2y \cos x$; for $y < 0$, $\partial_y f = 0$. At $y=0$, the one-sided derivatives are $\lim_{h \to 0^+} \frac{h^2 \cos x - 0}{h} = 0$ and $\lim_{h \to 0^-} \frac{0 - 0}{h} = 0$. So $\partial_y f$ exists everywhere and evaluates to $2y \cos x$ for $y \ge 0$ and $0$ for $y < 0$. These two pieces patch continuously at $y=0$.
  \item \textbf{False:} For $\partial_y f$ to be $C^1$, its own partial derivatives must be continuous. However, for $y>0$, $\partial_y(\partial_y f) = 2 \cos x$, while for $y<0$, it is $0$. As $y \to 0$, the upper half approaches $2 \cos x$, which does not match the lower half limit of $0$ unless $\cos x = 0$. Thus $\partial_{yy} f$ has jump discontinuities along the $x$-axis.
  \item \textbf{False:} Since $\partial_{yy}f$ is not continuous (from the previous part), $f$ is not twice continuously differentiable on $\mathbb{R}^2$.
  \item \textbf{True:} Along the $x$-axis ($y=0$), the function is $f(x,0) = 0$ everywhere. Therefore, the function restricted to the $x$-axis is identically zero, meaning $\partial_x f(x,0) = 0$ everywhere, and consequently its derivative with respect to $x$ again, $\partial_{xx} f(0,0)$, is also $0$.
\end{enumerate}
\end{exercisesolution}

\begin{exercisesolution}[Solution of \cref{ex:kobel_differentiable_single_choice}]
\label{sol:kobel_differentiable_single_choice}
% Generator: Claude Opus 5 (High)
The correct answer is \textbf{(c)}.
\begin{enumerate}[label=\textbf{(\alph*)}]
  \item \textbf{False}, and this is the misconception the chapter opens with. Existence of every
    partial derivative does not even force continuity, let alone differentiability:
    \cref{ex:partial_derivatives_not_continuous} exhibits a function whose two partials exist
    everywhere and which is discontinuous at the origin. Partial derivatives see only the
    coordinate directions, and differentiability is a statement about all directions at once.
  \item \textbf{False}, and the word doing the damage is \emph{arbitrary}. Compare the displayed
    limit with \cref{def:differentiable}: they are the same limit, except that the definition
    requires $L$ to be \emph{linear}. Drop that and the condition becomes almost vacuous --- take
    $L(h) := f(h)-f(0)$, which makes the numerator identically zero for every $f$ whatsoever. The
    linearity of $L$ is not a technical convenience; it is the entire content of the definition,
    since differentiability means precisely that $f$ is approximated by a linear map.
  \item \textbf{True}, and it is true for the unexciting reason that it asks for too much.
    Differentiability at $0$ is the existence of a linear $L$ with
    $f(h)-f(0)-L(h) = o(\lVert h\rVert)$. Here the error is required to be
    $o(\lVert h\rVert^2)$, which for small $h$ is a strictly stronger demand: if
    $\lVert f(h)-f(0)-L(h)\rVert/\lVert h\rVert^2 \to 0$, then
    \[\frac{\lVert f(h)-f(0)-L(h)\rVert}{\lVert h\rVert}
      = \lVert h\rVert \cdot \frac{\lVert f(h)-f(0)-L(h)\rVert}{\lVert h\rVert^2} \to 0\]
    as well. So $f$ is differentiable, with $Df_0 = L$.
  \item \textbf{False.} Bounded partial derivatives are no better than existing ones for this
    purpose --- the counterexample of \textbf{(a)} has partials that are bounded near the origin
    along each axis, and boundedness says nothing about the behaviour in between. What \emph{does}
    suffice is \emph{continuity} of the partials, which is
    \cref{thm:continuous_partials_imply_differentiable}; note that this is a genuinely stronger
    hypothesis than the one offered here, and that \cref{ex:differentiable_not_c1} shows it is not
    necessary either.
\end{enumerate}
\end{exercisesolution}
